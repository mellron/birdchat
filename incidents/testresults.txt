CREATE   VIEW [dbo].[vwPledgeTicketDtl]
(
    PldgCode,
    LocationID,
    LocationName,
    Cusip,
    Ticket,
    SecDescription,
    SecurityType,
    VariableRate,
    MaturityDate,
    USBNAFlag,
    OriginalFace,
    CurrentFace,
    MarketValue,
    OtherPledgedPct,
    OtherPldgMktValue,
    AcctCode,
    OrgUnitId,
    AssetClass1,
    AssetClass2,
    LstSegment,
    FixItFlag,
    UsedPct,
    AsOfDate,
    LastModifiedDt,
  Mktprc,
  parfact
)
AS
WITH Pledged_Used_Pct
AS (
   SELECT PldgCd,
          Ticket,
          sfkpid,
          SUM(Pctused) AS [Percentage],
          SUM(TicketDtlAmt) AS TicketDtlAmt,
          SUM(TicketHoldAmt) AS TicketHoldAmt,
          MAX(InsertDt) AS [InsertDt]
   FROM
   (
       SELECT sfkpcd.PldgCd,
              th.Ticket,
              sa.sfkpid,
              CAST(((ISNULL(CONVERT(FLOAT, tdtl.Amt) / NULLIF(CONVERT(FLOAT, th.Amt), 0), 0))) AS DECIMAL(38,20)) AS Pctused,
              tdtl.Amt TicketDtlAmt,
              th.Amt AS TicketHoldAmt,
              tdtl.InsertDt
       FROM TicketScrty vw
           INNER JOIN TicketHolding th
               ON vw.Ticket = th.Ticket
           INNER JOIN tblSfkpAcct sa1
               ON sa1.sfkpAcctID = th.sfkpAcctID
           LEFT JOIN TicketDtl tdtl
               ON tdtl.TicketHeldBlkId = th.TicketHeldBlk
           INNER JOIN tblSfkpPldgCd sfkpcd
               ON tdtl.SfkpPldgCdID = sfkpcd.SfkpPldgCdID
           INNER JOIN tblSfkpAcct sa
               ON sa.sfkpAcctID = sfkpcd.sfkpAcctID
       WHERE sa1.sfkpid = sa.sfkpid
   ) AS Tab
   GROUP BY PldgCd,
            Ticket,
            sfkpid
   ),
     Unpledged_Used_Pct
AS (SELECT 'Unpledged' AS PldgCd,
           vw.Ticket,
           sa1.sfkpid,
           (1 - ISNULL(TotalUsedPct, 0)) AS [Percentage],
           (th.Amt - ISNULL(UsedTicketDtlAmt, 0)) AS TicketDtlAmt,
           NULL AS Insertdt
    FROM TicketScrty vw
        INNER JOIN TicketHolding TH
            ON VW.Ticket = TH.Ticket
        INNER JOIN tblSfkpAcct sa1
            ON sa1.sfkpAcctID = th.sfkpAcctID
        LEFT OUTER JOIN
        (
            SELECT Ticket,
                   sfkpid,
                   SUM([Percentage]) AS TotalUsedPct,
                   SUM(TicketDtlAmt) AS UsedTicketDtlAmt
            FROM pledged_used_pct
            GROUP BY Ticket,
                     sfkpid
        ) AS Tab
            ON Tab.Ticket = vw.Ticket
               AND Tab.sfkpid = sa1.sfkpid
   ),
     All_Tickets
AS (SELECT *
    FROM
    (
        SELECT PldgCd,
               Ticket,
               sfkpid,
               [Percentage],
               TicketDtlAmt,
               (1 - [Percentage]) AS OtherPledgesPct,
               InsertDt
        FROM Pledged_Used_Pct
  WHERE [Percentage] > 0  AND TicketDtlAmt > 0
        UNION
        SELECT PldgCd,
               Ticket,
               sfkpid,
               [Percentage],
               TicketDtlAmt,
               (1 - [Percentage]) AS OtherPledgesPct,
               InsertDt
        FROM Unpledged_Used_Pct
        WHERE [Percentage] > 0  AND TicketDtlAmt > 0
    ) AS Tab
   ),
     AvailabilityReportCTE
AS (SELECT dtl.PldgCd AS PldgCode,
           ISNULL(csdtn.sfkpID, '') AS LocationId,
           ISNULL(csdtn.Descr, '') AS LocationName,
           TRIM(vw.Cusip) AS Cusip,
           vw.Ticket AS Ticket,
           vw.Descr AS SecDescription,
           vw.SecurityType,
           vw.VariableRate,
           vw.MatrDt AS MaturityDate,
           vw.USBNAFlag,
           TicketDtlAmt AS OriginalFace,
           th.currentface* ISNULL(dtl.[Percentage], 0) AS CurrentFace,
           th.Marketvalue* ISNULL(dtl.[Percentage], 0) AS MarketValue,
           ISNULL(dtl.OtherPledgesPct, 0) AS OtherPledgedPct,
           ((th.Amt - TicketDtlAmt) * ts.ParFact * ts.MktPrc ) AS OtherPldgMktValue,
           AcctCode,
           dbo.fnGetLegalEntityId(UPPER(vw.Portfolio)) AS OrgUnitId,
           tab.ALPCCategoryLevel1 AS AssetClass1,
           tab.ALPCCategoryLevel2 AS AssetClass2,
           '' AS LstSegment,
           vw.VariableRate AS FixItFlag,
           ISNULL(dtl.[Percentage], 0) AS UsedPct,
       th.ProcessDate as AsOfDate  ,
           dtl.InsertDt AS LastModifiedDt,
       ts.MktPrc,
       ts.ParFact
    FROM TicketScrty vw
        INNER JOIN TicketHolding th
            ON th.Ticket = vw.Ticket
    LEFT OUTER JOIN tblScrty ts ON ts.cusip = vw.cusip
        LEFT JOIN
        (
            SELECT ts.Ticket,
                   AC.ALPCCategoryLevel1,
                   AC.ALPCCategoryLevel2
            FROM TicketScrty ts
                INNER JOIN AssetClasses AC
                    ON CAST(TS.SecurityType AS INT) = CAST(AC.SecTypeCode AS INT)
                       AND CASE TRIM(ts.VariableRate)
                               WHEN '' THEN
                                   'F'
                               ELSE
                                   TRIM(ts.VariableRate)
                           END = TRIM(AC.FixFltFlag)
        ) AS tab
            ON tab.Ticket = vw.Ticket
        LEFT OUTER JOIN All_Tickets AS dtl
            ON dtl.Ticket = vw.Ticket
        LEFT OUTER JOIN TblSfkpAcct acct
            ON th.SfkpAcctID = acct.SfkpAcctID
        LEFT OUTER JOIN TblSfkpCstdn csdtn
            ON acct.SfkpID = csdtn.SfkpID
    WHERE dtl.SfkpID = acct.SfkpID
   )
SELECT *
FROM AvailabilityReportCTE
